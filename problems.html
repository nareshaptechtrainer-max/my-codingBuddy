🔍 Linear Search

O(n)
Walk the array left to right; return the index of the first match, or −1.
def linear_search(arr, target):
    for i in range(len(arr)):
        if arr[i] == target:
            return i
    return -1
int linearSearch(vector<int>& arr, int target) {
    for (int i = 0; i < arr.size(); i++) {
        if (arr[i] == target) return i;
    }
    return -1;
}
static int linearSearch(int[] arr, int target) {
    for (int i = 0; i < arr.length; i++) {
        if (arr[i] == target) return i;
    }
    return -1;
}
Time: O(n) · Space: O(1) — works on ANY array, sorted or not.

✂️ Binary Search

O(log n)needs sorted
Halve the search space every step. A million elements → ~20 checks.
def binary_search(arr, target):
    lo, hi = 0, len(arr) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1
int binarySearch(vector<int>& arr, int target) {
    int lo = 0, hi = arr.size() - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;  // overflow-safe!
        if (arr[mid] == target) return mid;
        else if (arr[mid] < target) lo = mid + 1;
        else hi = mid - 1;
    }
    return -1;
}
static int binarySearch(int[] arr, int target) {
    int lo = 0, hi = arr.length - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;  // overflow-safe!
        if (arr[mid] == target) return mid;
        else if (arr[mid] < target) lo = mid + 1;
        else hi = mid - 1;
    }
    return -1;
}
Time: O(log n) · Space: O(1) · Note the C++/Java trick lo + (hi−lo)/2 — plain (lo+hi)/2 can overflow int. Python ints never overflow.

🫧 Bubble Sort

O(n²)learning sort
Repeatedly swap adjacent out-of-order pairs; the biggest value "bubbles" to the end each pass.
def bubble_sort(arr):
    n = len(arr)
    for i in range(n - 1):
        swapped = False
        for j in range(n - 1 - i):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
                swapped = True
        if not swapped:      # already sorted → done
            break
void bubbleSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n - 1; i++) {
        bool swapped = false;
        for (int j = 0; j < n - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                swap(arr[j], arr[j + 1]);
                swapped = true;
            }
        }
        if (!swapped) break;   // already sorted
    }
}
static void bubbleSort(int[] arr) {
    int n = arr.length;
    for (int i = 0; i < n - 1; i++) {
        boolean swapped = false;
        for (int j = 0; j < n - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                int t = arr[j]; arr[j] = arr[j + 1]; arr[j + 1] = t;
                swapped = true;
            }
        }
        if (!swapped) break;   // already sorted
    }
}
Time: O(n²), best O(n) with the swapped flag · Space: O(1) · Real code uses sorted() / sort() / Arrays.sort() — O(n log n).

#️⃣ Hashing — Frequency Count

O(n)
The single most reused snippet in interviews: count how often each value appears.
def frequency(arr):
    counts = {}
    for x in arr:
        counts[x] = counts.get(x, 0) + 1
    return counts
# or: from collections import Counter
#     counts = Counter(arr)
unordered_map<int, int> frequency(vector<int>& arr) {
    unordered_map<int, int> counts;
    for (int x : arr)
        counts[x]++;          // auto-creates as 0
    return counts;
}
static Map<Integer, Integer> frequency(int[] arr) {
    Map<Integer, Integer> counts = new HashMap<>();
    for (int x : arr)
        counts.merge(x, 1, Integer::sum);
    return counts;
}
Time: O(n) · Space: O(k) distinct values · Unlocks: anagrams, duplicates, majority element, two sum.

🕸️ BFS & DFS on a Graph

O(V + E)
The two traversals that solve 90% of graph questions — full animations on the Graphs page.
from collections import deque

def bfs(graph, start):
    visited = {start}
    q = deque([start])
    while q:
        node = q.popleft()
        for nb in graph[node]:
            if nb not in visited:
                visited.add(nb)
                q.append(nb)

def dfs(graph, node, visited):
    visited.add(node)
    for nb in graph[node]:
        if nb not in visited:
            dfs(graph, nb, visited)
void bfs(vector<vector<int>>& g, int start) {
    vector<bool> visited(g.size(), false);
    queue<int> q;
    visited[start] = true; q.push(start);
    while (!q.empty()) {
        int node = q.front(); q.pop();
        for (int nb : g[node])
            if (!visited[nb]) {
                visited[nb] = true; q.push(nb);
            }
    }
}

void dfs(vector<vector<int>>& g, int node,
         vector<bool>& visited) {
    visited[node] = true;
    for (int nb : g[node])
        if (!visited[nb]) dfs(g, nb, visited);
}
static void bfs(List<List<Integer>> g, int start) {
    boolean[] visited = new boolean[g.size()];
    Deque<Integer> q = new ArrayDeque<>();
    visited[start] = true; q.offer(start);
    while (!q.isEmpty()) {
        int node = q.poll();
        for (int nb : g.get(node))
            if (!visited[nb]) {
                visited[nb] = true; q.offer(nb);
            }
    }
}

static void dfs(List<List<Integer>> g, int node,
                boolean[] visited) {
    visited[node] = true;
    for (int nb : g.get(node))
        if (!visited[nb]) dfs(g, nb, visited);
}
Both O(V + E) · BFS = shortest path (unweighted) · DFS = components, cycles, flood fill.

1 · Reverse a Number

EASY
1234 → 4321. No string conversion allowed.
💭 How to think: peel the last digit with % 10, glue it onto rev × 10, drop it with / 10. Repeat till zero.
def reverse_number(n):
    rev = 0
    while n > 0:
        rev = rev * 10 + n % 10
        n //= 10
    return rev
# reverse_number(1234) → 4321
int reverseNumber(int n) {
    int rev = 0;
    while (n > 0) {
        rev = rev * 10 + n % 10;
        n /= 10;
    }
    return rev;
}
static int reverseNumber(int n) {
    int rev = 0;
    while (n > 0) {
        rev = rev * 10 + n % 10;
        n /= 10;
    }
    return rev;
}
Time: O(digits) · Space: O(1)

2 · Palindrome Number

EASY
Is 12321 the same backwards? True. 123? False.
💭 How to think: reverse it with the previous trick, compare with the original. Done.
def is_palindrome(n):
    original, rev = n, 0
    while n > 0:
        rev = rev * 10 + n % 10
        n //= 10
    return original == rev
bool isPalindrome(int n) {
    int original = n;
    long long rev = 0;      // avoid overflow
    while (n > 0) {
        rev = rev * 10 + n % 10;
        n /= 10;
    }
    return original == rev;
}
static boolean isPalindrome(int n) {
    int original = n;
    long rev = 0;           // avoid overflow
    while (n > 0) {
        rev = rev * 10 + n % 10;
        n /= 10;
    }
    return original == rev;
}
Time: O(digits) · Space: O(1) · Note the long long/long in C++/Java — reversing can overflow int!

3 · Prime Check

EASY
Is n divisible only by 1 and itself?
💭 How to think: a factor pair always has one member ≤ √n — so testing up to √n is enough. 10⁹ needs only ~31,623 checks.
def is_prime(n):
    if n < 2:
        return False
    i = 2
    while i * i <= n:
        if n % i == 0:
            return False
        i += 1
    return True
bool isPrime(int n) {
    if (n < 2) return false;
    for (int i = 2; (long long)i * i <= n; i++)
        if (n % i == 0) return false;
    return true;
}
static boolean isPrime(int n) {
    if (n < 2) return false;
    for (int i = 2; (long) i * i <= n; i++)
        if (n % i == 0) return false;
    return true;
}
Time: O(√n) · Space: O(1)

🟡 Medium Problems

The placement core — each one demonstrates a reusable pattern

1 · Two Sum

MEDIUMhashingLC 1
Find indices of the two numbers adding to target. [2,7,11,15], target 9 → [0,1].
💭 How to think: for each number, the partner it NEEDS is target − x. Keep every number you've seen in a hash map — one lookup tells you if the partner already appeared.
def two_sum(nums, target):
    seen = {}                    # value → index
    for i, x in enumerate(nums):
        need = target - x
        if need in seen:
            return [seen[need], i]
        seen[x] = i
    return []
vector<int> twoSum(vector<int>& nums, int target) {
    unordered_map<int, int> seen;   // value → index
    for (int i = 0; i < nums.size(); i++) {
        int need = target - nums[i];
        if (seen.count(need))
            return {seen[need], i};
        seen[nums[i]] = i;
    }
    return {};
}
static int[] twoSum(int[] nums, int target) {
    Map<Integer, Integer> seen = new HashMap<>();
    for (int i = 0; i < nums.length; i++) {
        int need = target - nums[i];
        if (seen.containsKey(need))
            return new int[]{seen.get(need), i};
        seen.put(nums[i], i);
    }
    return new int[]{};
}
Time: O(n) · Space: O(n) — vs O(n²) brute force. The single most-asked interview question on earth.

2 · Maximum Subarray (Kadane)

MEDIUMkadaneLC 53
Largest sum of any contiguous run. [-2,1,-3,4,-1,2,1,-5,4] → 6 (from [4,-1,2,1]).
💭 How to think: at each element ask ONE question — extend the previous run, or start fresh here? A negative running sum can only hurt; drop it.
def max_subarray(nums):
    best = cur = nums[0]
    for x in nums[1:]:
        cur = max(x, cur + x)    # extend or restart
        best = max(best, cur)
    return best
int maxSubarray(vector<int>& nums) {
    int best = nums[0], cur = nums[0];
    for (int i = 1; i < nums.size(); i++) {
        cur = max(nums[i], cur + nums[i]);
        best = max(best, cur);
    }
    return best;
}
static int maxSubarray(int[] nums) {
    int best = nums[0], cur = nums[0];
    for (int i = 1; i < nums.length; i++) {
        cur = Math.max(nums[i], cur + nums[i]);
        best = Math.max(best, cur);
    }
    return best;
}
Time: O(n) · Space: O(1) · Animated step by step on the Arrays page.

3 · Remove Duplicates from Sorted Array

MEDIUMtwo pointersLC 26
In place, return the new length. [1,1,2,2,3] → 3 (array becomes [1,2,3,...]).
💭 How to think: slow pointer = "last kept position", fast pointer scans. A new value? Move it next to the kept ones.
def remove_duplicates(nums):
    if not nums:
        return 0
    slow = 0
    for fast in range(1, len(nums)):
        if nums[fast] != nums[slow]:
            slow += 1
            nums[slow] = nums[fast]
    return slow + 1
int removeDuplicates(vector<int>& nums) {
    if (nums.empty()) return 0;
    int slow = 0;
    for (int fast = 1; fast < nums.size(); fast++) {
        if (nums[fast] != nums[slow]) {
            slow++;
            nums[slow] = nums[fast];
        }
    }
    return slow + 1;
}
static int removeDuplicates(int[] nums) {
    if (nums.length == 0) return 0;
    int slow = 0;
    for (int fast = 1; fast < nums.length; fast++) {
        if (nums[fast] != nums[slow]) {
            slow++;
            nums[slow] = nums[fast];
        }

}

4 · Best Window of Size K

MEDIUMsliding windowLC 643
Maximum sum of any k consecutive elements.
💭 How to think: don't re-add k numbers per position! Slide: subtract the element leaving the window, add the one entering. Each slide is O(1).
def max_window_sum(nums, k):
    window = sum(nums[:k])
    best = window
    for i in range(k, len(nums)):
        window += nums[i] - nums[i - k]
        best = max(best, window)
    return best
int maxWindowSum(vector<int>& nums, int k) {
    int window = 0;
    for (int i = 0; i < k; i++) window += nums[i];
    int best = window;
    for (int i = k; i < nums.size(); i++) {
        window += nums[i] - nums[i - k];
        best = max(best, window);
    }
    return best;
}
static int maxWindowSum(int[] nums, int k) {
    int window = 0;
    for (int i = 0; i < k; i++) window += nums[i];
    int best = window;
    for (int i = k; i < nums.length; i++) {
        window += nums[i] - nums[i - k];
        best = Math.max(best, window);
    }
    return best;
}
Time: O(n) · Space: O(1) — vs O(n·k) recomputing. Animated on the Arrays page.

5 · Valid Anagram

MEDIUMhashingLC 242
Do "listen" and "silent" use exactly the same letters?
💭 How to think: same letters = same frequency counts. Count one string up, count the other down — everything should return to zero.
def is_anagram(s, t):
    if len(s) != len(t):
        return False
    counts = {}
    for ch in s:
        counts[ch] = counts.get(ch, 0) + 1
    for ch in t:
        if counts.get(ch, 0) == 0:
            return False
        counts[ch] -= 1
    return True
# one-liner: sorted(s) == sorted(t) — O(n log n)
bool isAnagram(string s, string t) {
    if (s.size() != t.size()) return false;
    int counts[26] = {0};
    for (char ch : s) counts[ch - 'a']++;
    for (char ch : t) {
        if (--counts[ch - 'a'] < 0) return false;
    }
    return true;
}
static boolean isAnagram(String s, String t) {
    if (s.length() != t.length()) return false;
    int[] counts = new int[26];
    for (char ch : s.toCharArray()) counts[ch - 'a']++;
    for (char ch : t.toCharArray()) {
        if (--counts[ch - 'a'] < 0) return false;
    }
    return true;
}
Time: O(n) · Space: O(1) (26 letters) · C++/Java use a plain 26-slot array as the "hash map" — a common micro-optimization.

6 · Move Zeroes

MEDIUMtwo pointersLC 283
Push all 0s to the end, keep the order of everything else. In place!
💭 How to think: slow pointer marks where the next NON-zero belongs. Fast pointer finds non-zeros and swaps them into place — zeros drift right automatically.
def move_zeroes(nums):
    slow = 0
    for fast in range(len(nums)):
        if nums[fast] != 0:
            nums[slow], nums[fast] = nums[fast], nums[slow]
            slow += 1
void moveZeroes(vector<int>& nums) {
    int slow = 0;
    for (int fast = 0; fast < nums.size(); fast++) {
        if (nums[fast] != 0) {
            swap(nums[slow], nums[fast]);
            slow++;
        }
    }
}
static void moveZeroes(int[] nums) {
    int slow = 0;
    for (int fast = 0; fast < nums.length; fast++) {
        if (nums[fast] != 0) {
            int t = nums[slow]; nums[slow] = nums[fast];
            nums[fast] = t;
            slow++;
        }
    }
}
Time: O(n) · Space: O(1)

1 · Trapping Rain Water

HARDtwo pointersLC 42
Bars of heights [0,1,0,2,1,0,1,3,2,1,2,1] — how much rain collects between them? → 6 units.
💭 How to think: water above a bar = min(tallest-to-left, tallest-to-right) − bar. Two pointers move inward from both ends; the side with the SMALLER max is safe to settle, because the other side is guaranteed taller.
def trap(height):
    l, r = 0, len(height) - 1
    left_max = right_max = water = 0
    while l < r:
        if height[l] < height[r]:
            left_max = max(left_max, height[l])
            water += left_max - height[l]
            l += 1
        else:
            right_max = max(right_max, height[r])
            water += right_max - height[r]
            r -= 1
    return water
int trap(vector<int>& height) {
    int l = 0, r = height.size() - 1;
    int leftMax = 0, rightMax = 0, water = 0;
    while (l < r) {
        if (height[l] < height[r]) {
            leftMax = max(leftMax, height[l]);
            water += leftMax - height[l];
            l++;
        } else {
            rightMax = max(rightMax, height[r]);
            water += rightMax - height[r];
            r--;
        }
    }
    return water;
}
static int trap(int[] height) {
    int l = 0, r = height.length - 1;
    int leftMax = 0, rightMax = 0, water = 0;
    while (l < r) {
        if (height[l] < height[r]) {
            leftMax = Math.max(leftMax, height[l]);
            water += leftMax - height[l];
            l++;
        } else {
            rightMax = Math.max(rightMax, height[r]);
            water += rightMax - height[r];
            r--;
        }
    }
    return water;
}
Time: O(n) · Space: O(1) — the elegant version of the O(n) space prefix-max approach.

2 · Longest Consecutive Sequence

HARDhashingLC 128
[100,4,200,1,3,2] → 4 (the run 1,2,3,4). Must be O(n) — sorting is banned.
💭 How to think: dump everything in a set. Only a number with NO left-neighbor (x−1 absent) can start a run — walk right from only those starts. Every number is touched at most twice.
def longest_consecutive(nums):
    s = set(nums)
    best = 0
    for x in s:
        if x - 1 not in s:          # run START only
            length = 1
            while x + length in s:
                length += 1
            best = max(best, length)
    return best
int longestConsecutive(vector<int>& nums) {
    unordered_set<int> s(nums.begin(), nums.end());
    int best = 0;
    for (int x : s) {
        if (!s.count(x - 1)) {        // run START only
            int length = 1;
            while (s.count(x + length)) length++;
            best = max(best, length);
        }
    }
    return best;
}
static int longestConsecutive(int[] nums) {
    Set<Integer> s = new HashSet<>();
    for (int x : nums) s.add(x);
    int best = 0;
    for (int x : s) {
        if (!s.contains(x - 1)) {     // run START only
            int length = 1;
            while (s.contains(x + length)) length++;
            best = Math.max(best, length);
        }
    }
    return best;
}
Time: O(n) · Space: O(n) · The "only start from run beginnings" insight is what makes it linear.

3 · Sort Colors (Dutch National Flag)

HARDthree pointersLC 75
Array of 0s, 1s, 2s — sort in ONE pass, no counting, no sort().
💭 How to think: three zones — low (0s), mid (scanning 1s), high (2s). See a 0? Swap to the low zone. See a 2? Swap to the high zone — but DON'T advance mid (the swapped-in value is unchecked!).
def sort_colors(nums):
    low, mid, high = 0, 0, len(nums) - 1
    while mid <= high:
        if nums[mid] == 0:
            nums[low], nums[mid] = nums[mid], nums[low]
            low += 1; mid += 1
        elif nums[mid] == 1:
            mid += 1
        else:  # 2
            nums[mid], nums[high] = nums[high], nums[mid]
            high -= 1        # mid stays!
void sortColors(vector<int>& nums) {
    int low = 0, mid = 0, high = nums.size() - 1;
    while (mid <= high) {
        if (nums[mid] == 0) {
            swap(nums[low], nums[mid]);
            low++; mid++;
        } else if (nums[mid] == 1) {
            mid++;
        } else {  // 2
            swap(nums[mid], nums[high]);
            high--;          // mid stays!
        }
    }
}
static void sortColors(int[] nums) {
    int low = 0, mid = 0, high = nums.length - 1;
    while (mid <= high) {
        if (nums[mid] == 0) {
            int t = nums[low]; nums[low] = nums[mid]; nums[mid] = t;
            low++; mid++;
        } else if (nums[mid] == 1) {
            mid++;
        } else {  // 2
            int t = nums[mid]; nums[mid] = nums[high]; nums[high] = t;
            high--;          // mid stays!
        }
    }
}
Time: O(n) one pass · Space: O(1) · Invented by Dijkstra. The "mid stays after the 2-swap" detail is what interviewers watch for.

4 · First Missing Positive

HARDindex as hashLC 41
[3,4,-1,1] → 2. O(n) time and O(1) extra space — the array must BE the hash table.
💭 How to think: the answer is in 1..n+1. Put every value v into its "home" index v−1 by swapping. Then the first index whose tenant is wrong reveals the missing number.
def first_missing_positive(nums):
    n = len(nums)
    for i in range(n):
        # keep swapping nums[i] to its home
        while 1 <= nums[i] <= n and \
              nums[nums[i] - 1] != nums[i]:
            home = nums[i] - 1
            nums[i], nums[home] = nums[home], nums[i]
    for i in range(n):
        if nums[i] != i + 1:
            return i + 1
    return n + 1
int firstMissingPositive(vector<int>& nums) {
    int n = nums.size();
    for (int i = 0; i < n; i++) {
        while (nums[i] >= 1 && nums[i] <= n
               && nums[nums[i] - 1] != nums[i]) {
            swap(nums[i], nums[nums[i] - 1]);
        }
    }
    for (int i = 0; i < n; i++)
        if (nums[i] != i + 1) return i + 1;
    return n + 1;
}
static int firstMissingPositive(int[] nums) {
    int n = nums.length;
    for (int i = 0; i < n; i++) {
        while (nums[i] >= 1 && nums[i] <= n
               && nums[nums[i] - 1] != nums[i]) {
            int home = nums[i] - 1;
            int t = nums[i]; nums[i] = nums[home]; nums[home] = t;
        }
    }
    for (int i = 0; i < n; i++)
        if (nums[i] != i + 1) return i + 1;
    return n + 1;
}
Time: O(n) (each value swapped to its home at most once) · Space: O(1) · "Cyclic sort" — the index-as-hash trick.
